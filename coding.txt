【BINARY SEARCH (O(log n))】
function binarySearch(arr, target):
    left = 0, right = arr.length - 1
    while left <= right:
        mid = left + (right - left) // 2  # Hindari overflow
        if arr[mid] == target: return mid
        elif arr[mid] < target: left = mid + 1
        else: right = mid - 1
    return -1

【QUICK SORT (O(n log n) average, O(n²) worst)】
function quickSort(arr, low, high):
    if low < high:
        pivotIndex = partition(arr, low, high)
        quickSort(arr, low, pivotIndex - 1)
        quickSort(arr, pivotIndex + 1, high)

function partition(arr, low, high):
    pivot = arr[high]  # pilih pivot element
    i = low - 1  # index of smaller element
    for j = low to high-1:
        if arr[j] <= pivot:
            i++
            swap(arr[i], arr[j])
    swap(arr[i+1], arr[high])
    return i+1

【DIJKSTRA (Shortest Path, O(E log V))】
function dijkstra(graph, source):
    dist[source] = 0
    priority_queue = {(0, source)}
    while queue not empty:
        current_dist, u = queue.pop()
        if current_dist > dist[u]: continue
        for v, weight in graph[u]:
            new_dist = dist[u] + weight
            if new_dist < dist[v]:
                dist[v] = new_dist
                queue.push((new_dist, v))
    return dist

【KNAPSACK DP (O(n·W))】
function knapsack(values, weights, capacity):
    n = len(values)
    dp = [[0] * (capacity+1) for _ in range(n+1)]
    for i from 1 to n:
        for w from 1 to capacity:
            if weights[i-1] <= w:
                dp[i][w] = max(dp[i-1][w],
                               values[i-1] + dp[i-1][w - weights[i-1]])
            else:
                dp[i][w] = dp[i-1][w]
    return dp[n][capacity]

【LONGEST COMMON SUBSEQUENCE (O(m·n))】
function LCS(text1, text2):
    m, n = len(text1), len(text2)
    dp = [[0]*(n+1) for _ in range(m+1)]
    for i from 1 to m:
        for j from 1 to n:
            if text1[i-1] == text2[j-1]:
                dp[i][j] = dp[i-1][j-1] + 1
            else:
                dp[i][j] = max(dp[i-1][j], dp[i][j-1])
    return dp[m][n]

【LEVENSHTEIN DISTANCE (Edit Distance)】
function editDistance(s1, s2):
    dp = [[0]*(len(s2)+1) for _ in range(len(s1)+1)]
    for i in range(len(s1)+1): dp[i][0] = i
    for j in range(len(s2)+1): dp[0][j] = j
    for i from 1 to len(s1):
        for j from 1 to len(s2):
            if s1[i-1] == s2[j-1]:
                dp[i][j] = dp[i-1][j-1]
            else:
                dp[i][j] = 1 + min(dp[i-1][j], dp[i][j-1], dp[i-1][j-1])
    return dp[len(s1)][len(s2)]

Kasus A: FULL TABLE SCAN
```sql
-- Slow (tanpa index)
SELECT * FROM orders WHERE customer_id = 12345;

-- Fix: tambah index
CREATE INDEX idx_orders_customer_id ON orders(customer_id);
-- Slow (temporary + filesort)
SELECT * FROM orders ORDER BY created_at DESC;

-- Fix: index covering
CREATE INDEX idx_orders_created_at ON orders(created_at DESC);
-- Slow (nested loop tanpa index)
SELECT o.*, c.name
FROM orders o
JOIN customers c ON o.customer_id = c.id
WHERE o.status = 'pending';

-- Fix: composite index
CREATE INDEX idx_orders_status_customer ON orders(status, customer_id);
CREATE INDEX idx_customers_id ON customers(id);
-- Slow (subquery dieksekusi per row)
SELECT * FROM products
WHERE category_id IN (SELECT id FROM categories WHERE active = 1);

-- Fast (JOIN)
SELECT p.* FROM products p
JOIN categories c ON p.category_id = c.id
WHERE c.active = 1;
-- Slow (offset scan 90,000 rows)
SELECT * FROM orders ORDER BY id LIMIT 10 OFFSET 90000;

-- Fast (keyset pagination / seek method)
SELECT * FROM orders WHERE id > 90000 ORDER BY id LIMIT 10;
